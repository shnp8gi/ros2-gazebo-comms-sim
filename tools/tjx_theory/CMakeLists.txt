cmake_minimum_required(VERSION 3.16)
project(tjx_theory_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenMP)
get_filename_component(PKG "${CMAKE_CURRENT_SOURCE_DIR}/../../src/comms_sim_pkg" ABSOLUTE)
enable_testing()
add_library(drone_link_core STATIC
  ${PKG}/src/drone_link.cpp ${PKG}/src/awgn.cpp ${PKG}/src/lab_phy.cpp
  ${PKG}/src/hrcp_ldpc.cpp ${PKG}/src/hrcp_codec.cpp ${PKG}/src/hrcp_modem.cpp
  ${PKG}/src/hrcp_mac.cpp ${PKG}/src/hrcp_pairnet.cpp ${PKG}/src/empirical_phy.cpp
  ${PKG}/src/tjx_theoretical_phy.cpp ${PKG}/src/hrcp_signal_monitor.cpp
  ${PKG}/src/antenna_pattern_parser.cpp ${PKG}/src/comms_calculator.cpp)
target_include_directories(drone_link_core PUBLIC ${PKG}/include)
target_link_libraries(drone_link_core PUBLIC Eigen3::Eigen yaml-cpp)
if(OpenMP_CXX_FOUND)
  target_link_libraries(drone_link_core PUBLIC OpenMP::OpenMP_CXX)
endif()
foreach(tool tjx_generate_lut tjx_theory_run initial_connection_sweep radio_profile)
  add_executable(${tool} ${PKG}/src/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE drone_link_core)
endforeach()
add_executable(asymmetric_link_profile ../asymmetric_link_profile.cpp)
target_link_libraries(asymmetric_link_profile PRIVATE drone_link_core)
foreach(test empirical_phy hrcp_rssi hrcp_modem hrcp_pairnet hrcp_codec hrcp_ldpc lab_phy awgn tjx_theoretical_phy)
  add_executable(${test}_tests ${PKG}/test/${test}_tests.cpp)
  target_link_libraries(${test}_tests PRIVATE drone_link_core)
  add_test(NAME ${test}_tests COMMAND ${test}_tests)
endforeach()
add_executable(drone_link_tests ${PKG}/test/drone_link_tests.cpp)
target_link_libraries(drone_link_tests PRIVATE drone_link_core)
add_test(NAME drone_link_tests COMMAND drone_link_tests ${PKG}/config)
add_executable(link_budget_tests ${PKG}/test/link_budget_tests.cpp)
target_link_libraries(link_budget_tests PRIVATE drone_link_core)
add_test(NAME link_budget_tests COMMAND link_budget_tests ${PKG}/config ${PKG}/test/fixtures/link_budget_workbook.json)
